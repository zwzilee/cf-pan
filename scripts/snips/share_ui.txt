/* 分享设置对话框：有效期 / 次数 / 提取码 / 每日次数 / 每日流量 */
function shareOptsDlg(cur = {}) {
  return new Promise((resolve) => {
    const mask = document.createElement('div');
    mask.className = 'mask';
    mask.innerHTML = `<div class="dlg" style="max-width:460px">
      <h3>分享设置</h3>
      <p class="mut" style="margin:0 0 12px">留空就是不限制</p>
      <div class="sgrid">
        <label>有效期（天）</label><input class="input" id="sdDays" type="number" min="1" max="365" value="${cur.days || 7}">
        <label>最多下载次数</label><input class="input" id="sdMax" type="number" min="1" placeholder="不限" value="${cur.max || ''}">
        <label>提取码</label><input class="input" id="sdCode" maxlength="16" placeholder="不填则不需要" value="${esc(cur.code || '')}">
        <label>每天最多次数</label><input class="input" id="sdDayMax" type="number" min="1" placeholder="不限" value="${cur.dayMax || ''}">
        <label>每天流量（MB）</label><input class="input" id="sdDayMB" type="number" min="1" placeholder="不限" value="${cur.dayMB || ''}">
      </div>
      <div class="foot"><button class="btn" id="sdNo">取消</button><button class="btn primary" id="sdYes">确定</button></div>
    </div>`;
    document.body.appendChild(mask);
    const close = (v) => { mask.remove(); resolve(v); };
    mask.querySelector('#sdNo').onclick = () => close(null);
    mask.onclick = (e) => { if (e.target === mask) close(null); };
    mask.querySelector('#sdYes').onclick = () => close({
      days: Math.max(1, Math.min(365, Number(mask.querySelector('#sdDays').value) || 7)),
      max: Number(mask.querySelector('#sdMax').value) || 0,
      code: mask.querySelector('#sdCode').value.trim(),
      dayMax: Number(mask.querySelector('#sdDayMax').value) || 0,
      dayMB: Number(mask.querySelector('#sdDayMB').value) || 0,
    });
    setTimeout(() => mask.querySelector('#sdDays').focus(), 30);
  });
}

async function shareFile(id) {
  const o = await shareOptsDlg({});
  if (!o) return;
  try {
    const r = await api(`/api/files/${id}/share`, {
      method: 'POST',
      body: JSON.stringify({ ttlSec: o.days * 86400, maxDownloads: o.max || undefined, code: o.code || undefined, dailyMax: o.dayMax || 0, dailyBytes: (o.dayMB || 0) * 1048576 }),
    });
    const url = location.origin + r.url;
    await navigator.clipboard.writeText(url).catch(() => {});
    toast(r.code ? `链接已复制，提取码 ${r.code}` : '分享链接已复制：' + url, 'ok');
  } catch (e) { toast(e.message, 'err'); }
}

async function shareEdit(jti) {
  const cur = state.shares.find((s) => s.jti === jti) || {};
  const o = await shareOptsDlg({
    days: cur.expiresAt ? Math.max(1, Math.round((new Date(cur.expiresAt) - Date.now()) / 86400000)) : 7,
    max: cur.max, code: cur.code, dayMax: cur.dayMax, dayMB: cur.byteMax ? Math.round(cur.byteMax / 1048576) : 0,
  });
  if (!o) return;
  try {
    await api('/api/shares/' + jti, {
      method: 'PATCH',
      body: JSON.stringify({ code: o.code, maxDownloads: o.max, dailyMax: o.dayMax, dailyBytes: o.dayMB * 1048576 }),
    });
    toast('已更新', 'ok'); renderShares();
  } catch (e) { toast(e.message, 'err'); }
}
